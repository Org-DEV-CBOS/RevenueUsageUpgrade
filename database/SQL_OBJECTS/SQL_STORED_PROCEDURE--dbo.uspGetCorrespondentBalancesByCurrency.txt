CREATE PROCEDURE dbo.uspGetCorrespondentBalancesByCurrency AS SELECT c.CurrencyId,c.CurrencyNameAr,c.CurrencyNameEn,c.Symbol,COALESCE(SUM(a.CurrentBalance),0)TotalBalance FROM dbo.Currencies c LEFT JOIN dbo.CorrespondentAccounts a ON a.CurrencyId=c.CurrencyId AND a.IsDeleted=0 WHERE c.IsDeleted=0 GROUP BY c.CurrencyId,c.CurrencyNameAr,c.CurrencyNameEn,c.Symbol;
