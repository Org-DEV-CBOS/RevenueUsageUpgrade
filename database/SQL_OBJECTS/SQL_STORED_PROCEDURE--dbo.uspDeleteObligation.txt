CREATE PROCEDURE dbo.uspDeleteObligation @ObligationId uniqueidentifier,@DeletedBy nvarchar(100) AS BEGIN IF EXISTS(SELECT 1 FROM dbo.ObligationPayments WHERE ObligationId=@ObligationId AND DeletedTime IS NULL) THROW 53001,'Obligation has payments.',1;UPDATE dbo.Obligations SET IsActive=0,DeletedTime=SYSUTCDATETIME(),DeletedBy=@DeletedBy WHERE ObligationId=@ObligationId;END;
