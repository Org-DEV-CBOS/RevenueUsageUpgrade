CREATE OR ALTER PROCEDURE dbo.uspAddObligationPayment @ObligationId uniqueidentifier,@CorrespondentAccountId uniqueidentifier,@PaymentDate date,@Amount decimal(19,4),@ReferenceNo nvarchar(100)=NULL,@Notes nvarchar(500)=NULL,@CreatedBy nvarchar(100) AS BEGIN SET XACT_ABORT ON;BEGIN TRAN;DECLARE @oc uniqueidentifier,@ac uniqueidentifier,@remaining decimal(19,4);SELECT @oc=CurrencyId,@remaining=TotalAmount-PaidAmount FROM dbo.Obligations WITH(UPDLOCK) WHERE ObligationId=@ObligationId AND IsActive=1 AND DeletedTime IS NULL;SELECT @ac=CurrencyId FROM dbo.CorrespondentAccounts WITH(UPDLOCK) WHERE CorrespondentAccountId=@CorrespondentAccountId AND IsActive=1 AND IsDeleted=0;IF @oc IS NULL THROW 53002,'Obligation not found.',1;IF @oc<>@ac THROW 53003,'Payment account currency mismatch.',1;IF @Amount>@remaining THROW 53004,'Payment exceeds remaining obligation.',1;UPDATE dbo.CorrespondentAccounts SET CurrentBalance=CurrentBalance-@Amount WHERE CorrespondentAccountId=@CorrespondentAccountId AND CurrentBalance>=@Amount;IF @@ROWCOUNT=0 THROW 53005,'Insufficient account balance.',1;INSERT dbo.ObligationPayments(ObligationId,CorrespondentAccountId,PaymentDate,Amount,ReferenceNo,Notes,CreatedBy)VALUES(@ObligationId,@CorrespondentAccountId,@PaymentDate,@Amount,@ReferenceNo,@Notes,@CreatedBy);UPDATE dbo.Obligations SET PaidAmount=PaidAmount+@Amount,IsActive=CASE WHEN PaidAmount+@Amount>=TotalAmount THEN 0 ELSE 1 END WHERE ObligationId=@ObligationId;COMMIT;END;
