CREATE PROCEDURE dbo.uspGetReserveSnapshots @StartDate date=NULL,@EndDate date=NULL AS SELECT ReserveSnapshotId,ReserveDate,GoldValue,CashInHand,Deposits,GoldValue+CashInHand+Deposits TotalValue,Notes FROM dbo.ReserveSnapshots WHERE DeletedTime IS NULL AND(@StartDate IS NULL OR ReserveDate>=@StartDate)AND(@EndDate IS NULL OR ReserveDate<=@EndDate)ORDER BY ReserveDate DESC,CreatedTime DESC;
