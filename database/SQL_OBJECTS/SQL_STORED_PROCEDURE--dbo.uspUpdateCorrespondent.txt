CREATE OR ALTER PROCEDURE dbo.uspUpdateCorrespondent @CorrespondentId uniqueidentifier,@CorrespondentCode nvarchar(50),@CorrespondentNameEn nvarchar(200),@CorrespondentNameAr nvarchar(200)=NULL,@CountryId uniqueidentifier=NULL,@IsActive bit,@ModifiedBy nvarchar(100)=NULL AS UPDATE dbo.Correspondents SET CorrespondentCode=@CorrespondentCode,CorrespondentNameEn=@CorrespondentNameEn,CorrespondentNameAr=@CorrespondentNameAr,CountryId=@CountryId,IsActive=@IsActive,ModifiedTime=SYSUTCDATETIME(),ModifiedBy=@ModifiedBy WHERE CorrespondentId=@CorrespondentId AND IsDeleted=0;
