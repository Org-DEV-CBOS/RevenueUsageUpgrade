CREATE PROCEDURE dbo.uspGetForeignReserveReport @StartDate date,@EndDate date AS SELECT r.ReserveDate ReportDate,COALESCE((SELECT SUM(CurrentBalance)FROM dbo.CorrespondentAccounts WHERE IsDeleted=0),0)CorrespondentBalancesUsd,r.CashInHand CashInHandUsd,r.GoldValue GoldValueUsd,r.Deposits DepositsUsd,COALESCE((SELECT SUM(Amount)FROM dbo.Resources WHERE ResourceDate BETWEEN @StartDate AND @EndDate AND DeletedTime IS NULL),0)ResourcesUsd,COALESCE((SELECT SUM(Amount)FROM dbo.Transfers WHERE TransferDate<DATEADD(day,1,@EndDate)AND TransferDate>=@StartDate AND TransferStatus='Confirmed'AND DeletedTime IS NULL),0)UsagesUsd,COALESCE((SELECT SUM(CurrentBalance)FROM dbo.CorrespondentAccounts WHERE IsDeleted=0),0)+r.CashInHand+r.GoldValue+r.Deposits GrandTotalUsd FROM dbo.ReserveSnapshots r WHERE r.ReserveDate BETWEEN @StartDate AND @EndDate AND r.DeletedTime IS NULL ORDER BY r.ReserveDate;
