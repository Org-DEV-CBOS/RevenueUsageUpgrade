CREATE PROCEDURE dbo.uspCreateBeneficiary @BeneficiaryCode nvarchar(50),@BeneficiaryNameEn nvarchar(200),@BeneficiaryNameAr nvarchar(200)=NULL,@CreatedBy nvarchar(100)=NULL AS BEGIN INSERT dbo.Beneficiaries(BeneficiaryCode,BeneficiaryNameEn,BeneficiaryNameAr,CreatedBy) OUTPUT inserted.BeneficiaryId VALUES(@BeneficiaryCode,@BeneficiaryNameEn,@BeneficiaryNameAr,@CreatedBy);END;
