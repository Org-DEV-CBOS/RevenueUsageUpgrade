CREATE OR ALTER PROCEDURE dbo.uspGetCurrencyBalances AS SELECT c.CurrencyId,c.CurrencyCode,c.CurrencyNameAr,c.CurrencyNameEn,COALESCE(SUM(a.OpeningBalance),0)TotalOpeningBalance,COALESCE((SELECT SUM(r.Amount) FROM dbo.Resources r JOIN dbo.CorrespondentAccounts ra ON ra.CorrespondentAccountId=r.CorrespondentAccountId WHERE ra.CurrencyId=c.CurrencyId AND r.DeletedTime IS NULL),0)TotalResources,COALESCE((SELECT SUM(x.Amount) FROM dbo.Coverages x JOIN dbo.CorrespondentAccounts xa ON xa.CorrespondentAccountId=x.FromCorrespondentAccountId WHERE xa.CurrencyId=c.CurrencyId AND x.DeletedTime IS NULL),0)TotalCoverageOut,COALESCE((SELECT SUM(x.Amount) FROM dbo.Coverages x JOIN dbo.CorrespondentAccounts xa ON xa.CorrespondentAccountId=x.ToCorrespondentAccountId WHERE xa.CurrencyId=c.CurrencyId AND x.DeletedTime IS NULL),0)TotalCoverageIn,COALESCE((SELECT SUM(t.Amount) FROM dbo.Transfers t JOIN dbo.CorrespondentAccounts ta ON ta.CorrespondentAccountId=t.CorrespondentAccountId WHERE ta.CurrencyId=c.CurrencyId AND t.TransferStatus='Confirmed' AND t.DeletedTime IS NULL),0)TotalConfirmedTransfers,COALESCE(SUM(a.CurrentBalance),0)CurrentBalance FROM dbo.Currencies c LEFT JOIN dbo.CorrespondentAccounts a ON a.CurrencyId=c.CurrencyId AND a.IsDeleted=0 WHERE c.IsDeleted=0 GROUP BY c.CurrencyId,c.CurrencyCode,c.CurrencyNameAr,c.CurrencyNameEn;
