CREATE PROCEDURE dbo.uspGetCorrespondents @ActiveOnly bit=1 AS SELECT c.CorrespondentId,c.CorrespondentCode,c.CorrespondentNameEn,c.CorrespondentNameAr,c.CountryId,k.CountryNameEn,k.CountryNameAr,c.IsActive FROM dbo.Correspondents c LEFT JOIN dbo.Countries k ON k.CountryId=c.CountryId WHERE c.IsDeleted=0 AND(@ActiveOnly=0 OR c.IsActive=1) ORDER BY c.CorrespondentNameEn;
