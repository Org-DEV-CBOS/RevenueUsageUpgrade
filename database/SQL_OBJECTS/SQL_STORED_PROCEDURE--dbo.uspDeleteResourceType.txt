CREATE PROCEDURE dbo.uspDeleteResourceType @ResourceTypeId uniqueidentifier,@DeletedBy nvarchar(100)=NULL AS BEGIN IF EXISTS(SELECT 1 FROM dbo.Resources WHERE ResourceTypeId=@ResourceTypeId AND DeletedTime IS NULL) THROW 51003,'Resource type has movements.',1;UPDATE dbo.ResourceTypes SET IsDeleted=1,IsActive=0,DeletedTime=SYSUTCDATETIME(),DeletedBy=@DeletedBy WHERE ResourceTypeId=@ResourceTypeId;END;
