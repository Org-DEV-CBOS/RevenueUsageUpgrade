CREATE OR ALTER PROCEDURE dbo.uspDeleteResource @ResourceId uniqueidentifier,@DeletedBy nvarchar(100) AS BEGIN SET XACT_ABORT ON;BEGIN TRAN;DECLARE @a uniqueidentifier,@v decimal(19,4);SELECT @a=CorrespondentAccountId,@v=Amount FROM dbo.Resources WITH(UPDLOCK) WHERE ResourceId=@ResourceId AND DeletedTime IS NULL;IF @a IS NULL THROW 52002,'Resource not found.',1;IF(SELECT CurrentBalance FROM dbo.CorrespondentAccounts WITH(UPDLOCK) WHERE CorrespondentAccountId=@a)<@v THROW 52003,'Insufficient balance to reverse resource.',1;UPDATE dbo.CorrespondentAccounts SET CurrentBalance=CurrentBalance-@v WHERE CorrespondentAccountId=@a;UPDATE dbo.Resources SET DeletedTime=SYSUTCDATETIME(),DeletedBy=@DeletedBy WHERE ResourceId=@ResourceId;COMMIT;END;
