CREATE PROCEDURE dbo.uspDeleteCurrency @CurrencyId uniqueidentifier,@DeletedBy nvarchar(100)=NULL AS BEGIN IF EXISTS(SELECT 1 FROM dbo.CorrespondentAccounts WHERE CurrencyId=@CurrencyId AND IsDeleted=0) THROW 51002,'Currency has accounts.',1;UPDATE dbo.Currencies SET IsDeleted=1,IsActive=0,DeletedTime=SYSUTCDATETIME(),DeletedBy=@DeletedBy WHERE CurrencyId=@CurrencyId;END;
