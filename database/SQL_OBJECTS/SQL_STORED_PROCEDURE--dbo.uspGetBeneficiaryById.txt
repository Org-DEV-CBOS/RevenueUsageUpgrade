CREATE PROCEDURE dbo.uspGetBeneficiaryById @BeneficiaryId uniqueidentifier AS SELECT BeneficiaryId,BeneficiaryCode,BeneficiaryNameEn,BeneficiaryNameAr,IsActive,CAST(CASE WHEN EXISTS(SELECT 1 FROM dbo.Transfers t WHERE t.BeneficiaryId=b.BeneficiaryId AND t.DeletedTime IS NULL) THEN 1 ELSE 0 END AS bit) HasMovements FROM dbo.Beneficiaries b WHERE BeneficiaryId=@BeneficiaryId AND IsDeleted=0;
