CREATE OR ALTER PROCEDURE dbo.uspGetDailyValuation @ValuationDate date=NULL AS SELECT r.ReserveSnapshotId DailyValuationId,r.ReserveDate ValuationDate,r.CashInHand CashInHandUsd,CAST(0 AS decimal(19,4))GoldOunces,CAST(0 AS decimal(19,4))GoldPricePerOunceUsd,r.GoldValue GoldValueUsd,r.Notes FROM dbo.ReserveSnapshots r WHERE r.DeletedTime IS NULL AND(@ValuationDate IS NULL OR r.ReserveDate=@ValuationDate)ORDER BY r.ReserveDate;
