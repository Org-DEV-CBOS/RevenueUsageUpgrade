CREATE OR ALTER PROCEDURE dbo.uspGetCorrespondentAccountStatement
    @CorrespondentAccountId uniqueidentifier,
    @StartDate datetime2 = NULL,
    @EndDate datetime2 = NULL
AS
BEGIN
    SET NOCOUNT ON;

    DECLARE @accountExists bit = 0;
    DECLARE @opening decimal(19, 4) = 0;

    SELECT @accountExists = 1,
           @opening = COALESCE(OpeningBalance, 0)
    FROM dbo.CorrespondentAccounts
    WHERE CorrespondentAccountId = @CorrespondentAccountId;

    /*
    Every movement carries two moments. EventDate is the business date the operation is
    booked under, which is what the caller filters on and what the statement groups by.
    EventTime is the moment the balance actually moved: when the row was written, or for
    a transfer, when it was confirmed, since a pending transfer has not touched the
    account yet. EventTime is what puts movements inside a day into their true order.
    */
    WITH e AS (
        SELECT CAST(ResourceDate AS datetime2) AS EventDate, CAST(CreatedTime AS datetime2) AS EventTime,
               'Resource' AS EventType,
               Amount AS AmountIn, CAST(0 AS decimal(19, 4)) AS AmountOut, Notes AS Notes
        FROM dbo.Resources
        WHERE CorrespondentAccountId = @CorrespondentAccountId AND DeletedTime IS NULL

        UNION ALL
        SELECT TransferDate, ConfirmedTime, 'Transfer', 0, Amount, Purpose
        FROM dbo.Transfers
        WHERE CorrespondentAccountId = @CorrespondentAccountId
          AND TransferStatus = 'Confirmed' AND DeletedTime IS NULL

        UNION ALL
        SELECT TransactionDate, CreatedTime, 'Coverage in', Amount, 0, Narration
        FROM dbo.Coverages
        WHERE ToCorrespondentAccountId = @CorrespondentAccountId AND DeletedTime IS NULL

        UNION ALL
        SELECT TransactionDate, CreatedTime, 'Coverage out', 0, Amount, Narration
        FROM dbo.Coverages
        WHERE FromCorrespondentAccountId = @CorrespondentAccountId AND DeletedTime IS NULL

        UNION ALL
        SELECT TransactionDate, CreatedTime, 'Deal in', ToAmount, 0, Narration
        FROM dbo.Deals
        WHERE ToCorrespondentAccountId = @CorrespondentAccountId AND DeletedTime IS NULL

        UNION ALL
        SELECT TransactionDate, CreatedTime, 'Deal out', 0, FromAmount, Narration
        FROM dbo.Deals
        WHERE FromCorrespondentAccountId = @CorrespondentAccountId AND DeletedTime IS NULL

        UNION ALL
        SELECT CAST(PaymentDate AS datetime2), CreatedTime, 'Obligation', 0, Amount, Notes
        FROM dbo.ObligationPayments
        WHERE CorrespondentAccountId = @CorrespondentAccountId AND DeletedTime IS NULL
    ),
    -- Movements before the window roll into the brought-forward balance so the
    -- running balance stays correct when the caller filters by date.
    broughtForward AS (
        SELECT COALESCE(SUM(AmountIn - AmountOut), 0) AS NetBefore
        FROM e
        WHERE @StartDate IS NOT NULL AND EventDate < @StartDate
    ),
    f AS (
        SELECT *
        FROM e
        WHERE (@StartDate IS NULL OR EventDate >= @StartDate)
          AND (@EndDate IS NULL OR EventDate < DATEADD(day, 1, CONVERT(date, @EndDate)))
    ),
    ledger AS (
        /*
        The statement opens with what the account was worth going into the window, so the
        running balance below has a stated starting point instead of an implied one.
        IsOpening marks it because it is not a movement: it carries no amounts and no
        time, and its date is the start of the window rather than the date of anything
        that happened.
        */
        SELECT
            0 AS SortOrder,
            CAST(1 AS bit) AS IsOpening,
            CAST(NULL AS date) AS DaySort,
            CAST(NULL AS datetime2) AS TimeSort,
            0 AS DirectionSort,
            CAST(@StartDate AS datetime2) AS EventDate,
            CAST(NULL AS datetime2) AS EventTime,
            CAST(NULL AS nvarchar(50)) AS EventType,
            CAST(0 AS decimal(19, 4)) AS AmountIn,
            CAST(0 AS decimal(19, 4)) AS AmountOut,
            @opening + bf.NetBefore AS RunningBalance,
            CAST(NULL AS nvarchar(500)) AS Notes
        FROM broughtForward bf
        WHERE @accountExists = 1

        UNION ALL

        /*
        Within a day, movements run in the order they actually hit the account. Rows
        predating the effective timestamps have none to sort by, so they fall back to
        their business date and post what came in ahead of what went out: the day closes
        on the same balance either way, but the account never shows a deficit it was
        never actually in. The running balance uses these same keys so it reads down the
        page in order.
        */
        SELECT
            1,
            CAST(0 AS bit),
            CONVERT(date, f.EventDate),
            COALESCE(f.EventTime, f.EventDate),
            CASE WHEN f.AmountIn > 0 THEN 0 ELSE 1 END,
            f.EventDate,
            f.EventTime,
            f.EventType,
            f.AmountIn,
            f.AmountOut,
            @opening + bf.NetBefore
                + SUM(f.AmountIn - f.AmountOut) OVER (
                    ORDER BY CONVERT(date, f.EventDate),
                             COALESCE(f.EventTime, f.EventDate),
                             CASE WHEN f.AmountIn > 0 THEN 0 ELSE 1 END,
                             f.EventType,
                             f.AmountOut,
                             f.AmountIn
                    ROWS UNBOUNDED PRECEDING),
            f.Notes
        FROM f
        CROSS JOIN broughtForward bf
    )
    SELECT
        IsOpening,
        EventDate,
        EventTime,
        EventType,
        AmountIn,
        AmountOut,
        RunningBalance,
        Notes
    FROM ledger
    ORDER BY SortOrder, DaySort, TimeSort, DirectionSort, EventType, AmountOut, AmountIn;
END;
