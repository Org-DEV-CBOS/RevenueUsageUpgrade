CREATE OR ALTER PROCEDURE dbo.uspGetCorrespondentAccountStatement @CorrespondentAccountId uniqueidentifier,@StartDate datetime2=NULL,@EndDate datetime2=NULL AS BEGIN DECLARE @opening decimal(19,4)=(SELECT OpeningBalance FROM dbo.CorrespondentAccounts WHERE CorrespondentAccountId=@CorrespondentAccountId);WITH e AS(SELECT CAST(ResourceDate AS datetime2)EventDate,'Resource'EventType,Amount AmountIn,CAST(0 AS decimal(19,4))AmountOut,Notes FROM dbo.Resources WHERE CorrespondentAccountId=@CorrespondentAccountId AND DeletedTime IS NULL UNION ALL SELECT TransferDate,'Transfer',0,Amount,Purpose FROM dbo.Transfers WHERE CorrespondentAccountId=@CorrespondentAccountId AND TransferStatus='Confirmed' AND DeletedTime IS NULL UNION ALL SELECT TransactionDate,'Coverage in',Amount,0,Narration FROM dbo.Coverages WHERE ToCorrespondentAccountId=@CorrespondentAccountId AND DeletedTime IS NULL UNION ALL SELECT TransactionDate,'Coverage out',0,Amount,Narration FROM dbo.Coverages WHERE FromCorrespondentAccountId=@CorrespondentAccountId AND DeletedTime IS NULL UNION ALL SELECT TransactionDate,'Deal in',ToAmount,0,Narration FROM dbo.Deals WHERE ToCorrespondentAccountId=@CorrespondentAccountId AND DeletedTime IS NULL UNION ALL SELECT TransactionDate,'Deal out',0,FromAmount,Narration FROM dbo.Deals WHERE FromCorrespondentAccountId=@CorrespondentAccountId AND DeletedTime IS NULL UNION ALL SELECT CAST(PaymentDate AS datetime2),'Obligation',0,Amount,Notes FROM dbo.ObligationPayments WHERE CorrespondentAccountId=@CorrespondentAccountId AND DeletedTime IS NULL),f AS(SELECT * FROM e WHERE(@StartDate IS NULL OR EventDate>=@StartDate)AND(@EndDate IS NULL OR EventDate<DATEADD(day,1,CONVERT(date,@EndDate))))SELECT EventDate,EventType,AmountIn,AmountOut,@opening+SUM(AmountIn-AmountOut)OVER(ORDER BY EventDate ROWS UNBOUNDED PRECEDING)RunningBalance,Notes FROM f ORDER BY EventDate;END;
