CREATE PROCEDURE dbo.uspUpdateResourceType @ResourceTypeId uniqueidentifier,@ResourceTypeCode nvarchar(50),@ResourceTypeNameEn nvarchar(100),@ResourceTypeNameAr nvarchar(100)=NULL,@IsActive bit,@ModifiedBy nvarchar(100)=NULL AS UPDATE dbo.ResourceTypes SET ResourceTypeCode=@ResourceTypeCode,ResourceTypeNameEn=@ResourceTypeNameEn,ResourceTypeNameAr=@ResourceTypeNameAr,IsActive=@IsActive,ModifiedTime=SYSUTCDATETIME(),ModifiedBy=@ModifiedBy WHERE ResourceTypeId=@ResourceTypeId AND IsDeleted=0;
