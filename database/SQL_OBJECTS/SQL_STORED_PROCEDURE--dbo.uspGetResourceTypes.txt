CREATE PROCEDURE dbo.uspGetResourceTypes @ActiveOnly bit=1 AS SELECT ResourceTypeId,ResourceTypeCode,ResourceTypeNameEn,ResourceTypeNameAr,IsActive,CAST(CASE WHEN EXISTS(SELECT 1 FROM dbo.Resources r WHERE r.ResourceTypeId=t.ResourceTypeId AND r.DeletedTime IS NULL) THEN 1 ELSE 0 END AS bit) HasMovements FROM dbo.ResourceTypes t WHERE IsDeleted=0 AND(@ActiveOnly=0 OR IsActive=1) ORDER BY ResourceTypeNameEn;
