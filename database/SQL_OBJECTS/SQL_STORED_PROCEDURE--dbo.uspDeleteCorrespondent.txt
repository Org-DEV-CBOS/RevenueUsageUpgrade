CREATE PROCEDURE dbo.uspDeleteCorrespondent @CorrespondentId uniqueidentifier,@DeletedBy nvarchar(100)=NULL AS BEGIN IF EXISTS(SELECT 1 FROM dbo.CorrespondentAccounts WHERE CorrespondentId=@CorrespondentId AND IsDeleted=0) THROW 51004,'Correspondent has accounts.',1;UPDATE dbo.Correspondents SET IsDeleted=1,IsActive=0,DeletedTime=SYSUTCDATETIME(),DeletedBy=@DeletedBy WHERE CorrespondentId=@CorrespondentId;END;
