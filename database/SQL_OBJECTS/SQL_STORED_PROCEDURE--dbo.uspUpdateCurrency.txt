CREATE PROCEDURE dbo.uspUpdateCurrency @CurrencyId uniqueidentifier,@CurrencyCode char(3),@CurrencyNameEn nvarchar(100),@CurrencyNameAr nvarchar(100)=NULL,@Symbol nvarchar(10)=NULL,@DecimalPlaces tinyint=2,@IsActive bit,@ModifiedBy nvarchar(100)=NULL AS UPDATE dbo.Currencies SET CurrencyCode=UPPER(@CurrencyCode),CurrencyNameEn=@CurrencyNameEn,CurrencyNameAr=@CurrencyNameAr,Symbol=@Symbol,DecimalPlaces=@DecimalPlaces,IsActive=@IsActive,ModifiedTime=SYSUTCDATETIME(),ModifiedBy=@ModifiedBy WHERE CurrencyId=@CurrencyId AND IsDeleted=0;
