CREATE PROCEDURE dbo.uspConfirmTransfer @TransferId uniqueidentifier,@ConfirmedBy nvarchar(100) AS BEGIN SET XACT_ABORT ON;BEGIN TRAN;DECLARE @a uniqueidentifier,@v decimal(19,4),@s nvarchar(20);SELECT @a=CorrespondentAccountId,@v=Amount,@s=TransferStatus FROM dbo.Transfers WITH(UPDLOCK) WHERE TransferId=@TransferId AND DeletedTime IS NULL;IF @s<>'Pending' THROW 52010,'Only pending transfers can be confirmed.',1;UPDATE dbo.CorrespondentAccounts WITH(UPDLOCK) SET CurrentBalance=CurrentBalance-@v WHERE CorrespondentAccountId=@a AND CurrentBalance>=@v AND IsActive=1 AND IsDeleted=0;IF @@ROWCOUNT=0 THROW 52011,'Insufficient account balance.',1;UPDATE dbo.Transfers SET TransferStatus='Confirmed',ConfirmedTime=SYSUTCDATETIME(),ConfirmedBy=@ConfirmedBy WHERE TransferId=@TransferId;COMMIT;END;
