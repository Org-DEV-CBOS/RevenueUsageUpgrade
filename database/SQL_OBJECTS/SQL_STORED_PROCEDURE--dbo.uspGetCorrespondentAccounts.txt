CREATE PROCEDURE dbo.uspGetCorrespondentAccounts @CorrespondentId uniqueidentifier=NULL,@CurrencyId uniqueidentifier=NULL,@ActiveOnly bit=1 AS SELECT a.CorrespondentAccountId,a.CorrespondentId,c.CorrespondentNameEn,c.CorrespondentNameAr,a.CurrencyId,u.CurrencyCode,u.CurrencyNameEn,u.CurrencyNameAr,a.AccountNumber,a.OpeningBalance,a.CurrentBalance,a.IsActive,CAST(CASE WHEN EXISTS(SELECT 1 FROM dbo.Resources r WHERE r.CorrespondentAccountId=a.CorrespondentAccountId AND r.DeletedTime IS NULL) OR EXISTS(SELECT 1 FROM dbo.Transfers t WHERE t.CorrespondentAccountId=a.CorrespondentAccountId AND t.DeletedTime IS NULL) THEN 1 ELSE 0 END AS bit) HasMovements FROM dbo.CorrespondentAccounts a JOIN dbo.Correspondents c ON c.CorrespondentId=a.CorrespondentId JOIN dbo.Currencies u ON u.CurrencyId=a.CurrencyId WHERE a.IsDeleted=0 AND(@CorrespondentId IS NULL OR a.CorrespondentId=@CorrespondentId)AND(@CurrencyId IS NULL OR a.CurrencyId=@CurrencyId)AND(@ActiveOnly=0 OR a.IsActive=1) ORDER BY c.CorrespondentNameEn,u.CurrencyCode;
