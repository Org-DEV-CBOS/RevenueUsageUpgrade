CREATE OR ALTER PROCEDURE dbo.uspDeleteCorrespondentAccount @CorrespondentAccountId uniqueidentifier,@DeletedBy nvarchar(100)=NULL AS BEGIN IF EXISTS(SELECT 1 FROM dbo.Resources WHERE CorrespondentAccountId=@CorrespondentAccountId AND DeletedTime IS NULL) OR EXISTS(SELECT 1 FROM dbo.Transfers WHERE CorrespondentAccountId=@CorrespondentAccountId AND DeletedTime IS NULL) OR EXISTS(SELECT 1 FROM dbo.Coverages WHERE(FromCorrespondentAccountId=@CorrespondentAccountId OR ToCorrespondentAccountId=@CorrespondentAccountId)AND DeletedTime IS NULL) OR EXISTS(SELECT 1 FROM dbo.Deals WHERE(FromCorrespondentAccountId=@CorrespondentAccountId OR ToCorrespondentAccountId=@CorrespondentAccountId)AND DeletedTime IS NULL) THROW 51006,'Account has movements.',1;UPDATE dbo.CorrespondentAccounts SET IsDeleted=1,IsActive=0,DeletedTime=SYSUTCDATETIME(),DeletedBy=@DeletedBy WHERE CorrespondentAccountId=@CorrespondentAccountId;END;
