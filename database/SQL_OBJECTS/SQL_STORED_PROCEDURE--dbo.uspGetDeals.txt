CREATE PROCEDURE dbo.uspGetDeals @CorrespondentAccountId uniqueidentifier=NULL,@StartDate datetime2=NULL,@EndDate datetime2=NULL AS SELECT d.DealId,d.FromCorrespondentAccountId,fc.CorrespondentNameEn FromCorrespondentName,fu.CurrencyCode FromCurrencyCode,d.ToCorrespondentAccountId,tc.CorrespondentNameEn ToCorrespondentName,tu.CurrencyCode ToCurrencyCode,d.FromAmount,d.ExchangeRate,d.ToAmount,d.ReferenceNo,d.Narration,d.TransactionDate FROM dbo.Deals d JOIN dbo.CorrespondentAccounts fa ON fa.CorrespondentAccountId=d.FromCorrespondentAccountId JOIN dbo.Correspondents fc ON fc.CorrespondentId=fa.CorrespondentId JOIN dbo.Currencies fu ON fu.CurrencyId=fa.CurrencyId JOIN dbo.CorrespondentAccounts ta ON ta.CorrespondentAccountId=d.ToCorrespondentAccountId JOIN dbo.Correspondents tc ON tc.CorrespondentId=ta.CorrespondentId JOIN dbo.Currencies tu ON tu.CurrencyId=ta.CurrencyId WHERE d.DeletedTime IS NULL AND(@CorrespondentAccountId IS NULL OR d.FromCorrespondentAccountId=@CorrespondentAccountId OR d.ToCorrespondentAccountId=@CorrespondentAccountId)AND(@StartDate IS NULL OR d.TransactionDate>=@StartDate)AND(@EndDate IS NULL OR d.TransactionDate<DATEADD(day,1,CONVERT(date,@EndDate)))ORDER BY d.TransactionDate DESC;
