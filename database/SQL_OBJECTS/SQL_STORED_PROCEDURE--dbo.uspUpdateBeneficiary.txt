CREATE OR ALTER PROCEDURE dbo.uspUpdateBeneficiary @BeneficiaryId uniqueidentifier,@BeneficiaryCode nvarchar(50),@BeneficiaryNameEn nvarchar(200),@BeneficiaryNameAr nvarchar(200)=NULL,@IsActive bit,@ModifiedBy nvarchar(100)=NULL AS UPDATE dbo.Beneficiaries SET BeneficiaryCode=@BeneficiaryCode,BeneficiaryNameEn=@BeneficiaryNameEn,BeneficiaryNameAr=@BeneficiaryNameAr,IsActive=@IsActive,ModifiedTime=SYSUTCDATETIME(),ModifiedBy=@ModifiedBy WHERE BeneficiaryId=@BeneficiaryId AND IsDeleted=0;
