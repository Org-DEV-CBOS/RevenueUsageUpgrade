CREATE OR ALTER PROCEDURE dbo.uspGetCurrencies @ActiveOnly bit=1 AS SELECT CurrencyId,CurrencyCode,CurrencyNameEn,CurrencyNameAr,Symbol,DecimalPlaces,IsActive,CAST(CASE WHEN EXISTS(SELECT 1 FROM dbo.CorrespondentAccounts a WHERE a.CurrencyId=c.CurrencyId AND a.IsDeleted=0) THEN 1 ELSE 0 END AS bit) HasMovements FROM dbo.Currencies c WHERE IsDeleted=0 AND(@ActiveOnly=0 OR IsActive=1) ORDER BY CurrencyCode;
