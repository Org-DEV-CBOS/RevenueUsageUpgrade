CREATE OR ALTER PROCEDURE dbo.uspCreateDeal @FromCorrespondentAccountId uniqueidentifier,@ToCorrespondentAccountId uniqueidentifier,@FromAmount decimal(19,4),@ExchangeRate decimal(19,8),@ReferenceNo nvarchar(100)=NULL,@Narration nvarchar(500)=NULL,@TransactionDate datetime2,@CreatedBy nvarchar(100) AS BEGIN SET XACT_ABORT ON;BEGIN TRAN;DECLARE @fc uniqueidentifier,@tc uniqueidentifier,@to decimal(19,4)=ROUND(@FromAmount*@ExchangeRate,4);SELECT @fc=CurrencyId FROM dbo.CorrespondentAccounts WITH(UPDLOCK) WHERE CorrespondentAccountId=@FromCorrespondentAccountId AND IsActive=1 AND IsDeleted=0;SELECT @tc=CurrencyId FROM dbo.CorrespondentAccounts WITH(UPDLOCK) WHERE CorrespondentAccountId=@ToCorrespondentAccountId AND IsActive=1 AND IsDeleted=0;IF @fc IS NULL OR @tc IS NULL OR @fc=@tc THROW 52030,'Deal requires active accounts in different currencies.',1;UPDATE dbo.CorrespondentAccounts SET CurrentBalance=CurrentBalance-@FromAmount WHERE CorrespondentAccountId=@FromCorrespondentAccountId AND CurrentBalance>=@FromAmount;IF @@ROWCOUNT=0 THROW 52031,'Insufficient source balance.',1;UPDATE dbo.CorrespondentAccounts SET CurrentBalance=CurrentBalance+@to WHERE CorrespondentAccountId=@ToCorrespondentAccountId;DECLARE @id uniqueidentifier=NEWID();INSERT dbo.Deals(DealId,FromCorrespondentAccountId,ToCorrespondentAccountId,FromAmount,ExchangeRate,ReferenceNo,Narration,TransactionDate,CreatedBy)VALUES(@id,@FromCorrespondentAccountId,@ToCorrespondentAccountId,@FromAmount,@ExchangeRate,@ReferenceNo,@Narration,@TransactionDate,@CreatedBy);COMMIT;SELECT @id;END;
