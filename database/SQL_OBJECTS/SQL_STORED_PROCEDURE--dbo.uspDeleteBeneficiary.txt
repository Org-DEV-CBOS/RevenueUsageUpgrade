CREATE OR ALTER PROCEDURE dbo.uspDeleteBeneficiary @BeneficiaryId uniqueidentifier,@DeletedBy nvarchar(100)=NULL AS BEGIN IF EXISTS(SELECT 1 FROM dbo.Transfers WHERE BeneficiaryId=@BeneficiaryId AND DeletedTime IS NULL) THROW 51001,'Beneficiary has movements.',1;UPDATE dbo.Beneficiaries SET IsDeleted=1,IsActive=0,DeletedTime=SYSUTCDATETIME(),DeletedBy=@DeletedBy WHERE BeneficiaryId=@BeneficiaryId;END;
