CREATE OR ALTER PROCEDURE dbo.uspGetCurrencyStatement @CurrencyId uniqueidentifier,@AsOfDate datetime2 AS SELECT a.CorrespondentAccountId,a.AccountNumber,c.CorrespondentNameEn AccountName,a.CurrentBalance,COALESCE((SELECT SUM(r.Amount)FROM dbo.Resources r WHERE r.CorrespondentAccountId=a.CorrespondentAccountId AND r.ResourceDate<=@AsOfDate AND r.DeletedTime IS NULL),0)TotalResources,COALESCE((SELECT SUM(x.Amount)FROM dbo.Coverages x WHERE x.ToCorrespondentAccountId=a.CorrespondentAccountId AND x.TransactionDate<=@AsOfDate AND x.DeletedTime IS NULL),0)TotalCoverageIn,COALESCE((SELECT SUM(x.Amount)FROM dbo.Coverages x WHERE x.FromCorrespondentAccountId=a.CorrespondentAccountId AND x.TransactionDate<=@AsOfDate AND x.DeletedTime IS NULL),0)TotalCoverageOut,COALESCE((SELECT SUM(t.Amount)FROM dbo.Transfers t WHERE t.CorrespondentAccountId=a.CorrespondentAccountId AND t.TransferDate<=@AsOfDate AND t.TransferStatus='Confirmed' AND t.DeletedTime IS NULL),0)TotalConfirmedTransfers,a.CurrentBalance NetBalance FROM dbo.CorrespondentAccounts a JOIN dbo.Correspondents c ON c.CorrespondentId=a.CorrespondentId WHERE a.CurrencyId=@CurrencyId AND a.IsDeleted=0;
