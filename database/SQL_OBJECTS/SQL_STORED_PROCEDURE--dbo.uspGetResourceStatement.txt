CREATE OR ALTER PROCEDURE dbo.uspGetResourceStatement @CorrespondentAccountId uniqueidentifier,@StartDate date=NULL,@EndDate date=NULL AS SELECT r.ResourceId,r.ResourceDate,t.ResourceTypeNameEn ResourceType,r.Amount,CAST(NULL AS nvarchar(100))ReferenceNo,r.Notes FROM dbo.Resources r JOIN dbo.ResourceTypes t ON t.ResourceTypeId=r.ResourceTypeId WHERE r.CorrespondentAccountId=@CorrespondentAccountId AND r.DeletedTime IS NULL AND(@StartDate IS NULL OR r.ResourceDate>=@StartDate)AND(@EndDate IS NULL OR r.ResourceDate<=@EndDate)ORDER BY r.ResourceDate,r.CreatedTime;
