CREATE OR ALTER PROCEDURE dbo.uspCreateCoverage @FromCorrespondentAccountId uniqueidentifier,@ToCorrespondentAccountId uniqueidentifier,@Amount decimal(19,4),@ReferenceNo nvarchar(100)=NULL,@Narration nvarchar(500)=NULL,@TransactionDate datetime2,@CreatedBy nvarchar(100) AS BEGIN SET XACT_ABORT ON;BEGIN TRAN;DECLARE @fc uniqueidentifier,@tc uniqueidentifier;SELECT @fc=CurrencyId FROM dbo.CorrespondentAccounts WITH(UPDLOCK) WHERE CorrespondentAccountId=@FromCorrespondentAccountId AND IsActive=1 AND IsDeleted=0;SELECT @tc=CurrencyId FROM dbo.CorrespondentAccounts WITH(UPDLOCK) WHERE CorrespondentAccountId=@ToCorrespondentAccountId AND IsActive=1 AND IsDeleted=0;IF @fc IS NULL OR @tc IS NULL OR @fc<>@tc OR @FromCorrespondentAccountId=@ToCorrespondentAccountId THROW 52020,'Coverage requires distinct active accounts in the same currency.',1;UPDATE dbo.CorrespondentAccounts SET CurrentBalance=CurrentBalance-@Amount WHERE CorrespondentAccountId=@FromCorrespondentAccountId AND CurrentBalance>=@Amount;IF @@ROWCOUNT=0 THROW 52021,'Insufficient source balance.',1;UPDATE dbo.CorrespondentAccounts SET CurrentBalance=CurrentBalance+@Amount WHERE CorrespondentAccountId=@ToCorrespondentAccountId;DECLARE @id uniqueidentifier=NEWID();INSERT dbo.Coverages(CoverageId,FromCorrespondentAccountId,ToCorrespondentAccountId,Amount,ReferenceNo,Narration,TransactionDate,CreatedBy)VALUES(@id,@FromCorrespondentAccountId,@ToCorrespondentAccountId,@Amount,@ReferenceNo,@Narration,@TransactionDate,@CreatedBy);COMMIT;SELECT @id;END;
