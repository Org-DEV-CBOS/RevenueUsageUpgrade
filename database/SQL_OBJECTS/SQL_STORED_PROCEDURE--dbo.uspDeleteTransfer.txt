CREATE PROCEDURE dbo.uspDeleteTransfer @TransferId uniqueidentifier,@DeletedBy nvarchar(100) AS BEGIN SET XACT_ABORT ON;BEGIN TRAN;DECLARE @a uniqueidentifier,@v decimal(19,4),@s nvarchar(20);SELECT @a=CorrespondentAccountId,@v=Amount,@s=TransferStatus FROM dbo.Transfers WITH(UPDLOCK) WHERE TransferId=@TransferId AND DeletedTime IS NULL;IF @a IS NULL THROW 52012,'Transfer not found.',1;IF @s='Confirmed' UPDATE dbo.CorrespondentAccounts SET CurrentBalance=CurrentBalance+@v WHERE CorrespondentAccountId=@a;UPDATE dbo.Transfers SET DeletedTime=SYSUTCDATETIME(),DeletedBy=@DeletedBy WHERE TransferId=@TransferId;COMMIT;END;
