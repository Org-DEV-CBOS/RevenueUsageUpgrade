CREATE OR ALTER PROCEDURE dbo.uspDeleteObligationPayment @ObligationPaymentId uniqueidentifier,@DeletedBy nvarchar(100) AS BEGIN SET XACT_ABORT ON;BEGIN TRAN;DECLARE @o uniqueidentifier,@a uniqueidentifier,@v decimal(19,4);SELECT @o=ObligationId,@a=CorrespondentAccountId,@v=Amount FROM dbo.ObligationPayments WITH(UPDLOCK) WHERE ObligationPaymentId=@ObligationPaymentId AND DeletedTime IS NULL;IF @o IS NULL THROW 53006,'Payment not found.',1;UPDATE dbo.CorrespondentAccounts SET CurrentBalance=CurrentBalance+@v WHERE CorrespondentAccountId=@a;UPDATE dbo.Obligations SET PaidAmount=PaidAmount-@v,IsActive=1 WHERE ObligationId=@o;UPDATE dbo.ObligationPayments SET DeletedTime=SYSUTCDATETIME(),DeletedBy=@DeletedBy WHERE ObligationPaymentId=@ObligationPaymentId;COMMIT;END;
