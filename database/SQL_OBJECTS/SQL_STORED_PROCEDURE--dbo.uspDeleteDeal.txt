CREATE PROCEDURE dbo.uspDeleteDeal @DealId uniqueidentifier,@DeletedBy nvarchar(100) AS BEGIN SET XACT_ABORT ON;BEGIN TRAN;DECLARE @f uniqueidentifier,@t uniqueidentifier,@fv decimal(19,4),@tv decimal(19,4);SELECT @f=FromCorrespondentAccountId,@t=ToCorrespondentAccountId,@fv=FromAmount,@tv=ToAmount FROM dbo.Deals WITH(UPDLOCK) WHERE DealId=@DealId AND DeletedTime IS NULL;IF @f IS NULL THROW 52032,'Deal not found.',1;UPDATE dbo.CorrespondentAccounts SET CurrentBalance=CurrentBalance-@tv WHERE CorrespondentAccountId=@t AND CurrentBalance>=@tv;IF @@ROWCOUNT=0 THROW 52033,'Destination balance is insufficient to reverse deal.',1;UPDATE dbo.CorrespondentAccounts SET CurrentBalance=CurrentBalance+@fv WHERE CorrespondentAccountId=@f;UPDATE dbo.Deals SET DeletedTime=SYSUTCDATETIME(),DeletedBy=@DeletedBy WHERE DealId=@DealId;COMMIT;END;
