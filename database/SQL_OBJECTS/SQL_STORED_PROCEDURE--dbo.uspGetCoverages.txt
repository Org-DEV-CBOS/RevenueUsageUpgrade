CREATE OR ALTER PROCEDURE dbo.uspGetCoverages @CorrespondentAccountId uniqueidentifier=NULL,@StartDate datetime2=NULL,@EndDate datetime2=NULL AS SELECT x.CoverageId,x.FromCorrespondentAccountId,fc.CorrespondentNameEn FromCorrespondentName,x.ToCorrespondentAccountId,tc.CorrespondentNameEn ToCorrespondentName,fa.CurrencyId,u.CurrencyCode,x.Amount,x.ReferenceNo,x.Narration,x.TransactionDate FROM dbo.Coverages x JOIN dbo.CorrespondentAccounts fa ON fa.CorrespondentAccountId=x.FromCorrespondentAccountId JOIN dbo.Correspondents fc ON fc.CorrespondentId=fa.CorrespondentId JOIN dbo.CorrespondentAccounts ta ON ta.CorrespondentAccountId=x.ToCorrespondentAccountId JOIN dbo.Correspondents tc ON tc.CorrespondentId=ta.CorrespondentId JOIN dbo.Currencies u ON u.CurrencyId=fa.CurrencyId WHERE x.DeletedTime IS NULL AND(@CorrespondentAccountId IS NULL OR x.FromCorrespondentAccountId=@CorrespondentAccountId OR x.ToCorrespondentAccountId=@CorrespondentAccountId)AND(@StartDate IS NULL OR x.TransactionDate>=@StartDate)AND(@EndDate IS NULL OR x.TransactionDate<DATEADD(day,1,CONVERT(date,@EndDate)))ORDER BY x.TransactionDate DESC;
