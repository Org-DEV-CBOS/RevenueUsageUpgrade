CREATE PROCEDURE dbo.uspRejectTransfer @TransferId uniqueidentifier,@RejectedBy nvarchar(100),@RejectReason nvarchar(500)=NULL AS UPDATE dbo.Transfers SET TransferStatus='Rejected',RejectedTime=SYSUTCDATETIME(),RejectedBy=@RejectedBy,RejectReason=@RejectReason WHERE TransferId=@TransferId AND TransferStatus='Pending' AND DeletedTime IS NULL;
