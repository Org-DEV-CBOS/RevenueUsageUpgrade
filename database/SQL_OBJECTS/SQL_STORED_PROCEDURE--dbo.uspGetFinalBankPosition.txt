CREATE PROCEDURE dbo.uspGetFinalBankPosition @PositionDate date AS BEGIN DECLARE @cash decimal(19,4)=0,@gold decimal(19,4)=0;SELECT TOP(1)@cash=CashInHand,@gold=GoldValue FROM dbo.ReserveSnapshots WHERE ReserveDate<=@PositionDate AND DeletedTime IS NULL ORDER BY ReserveDate DESC,CreatedTime DESC;DECLARE @corr decimal(19,4)=COALESCE((SELECT SUM(CurrentBalance)FROM dbo.CorrespondentAccounts WHERE IsDeleted=0),0);SELECT @PositionDate PositionDate,@cash CashInHandUsd,@gold GoldValueUsd,@corr TotalCorrespondentBalancesUsd,@cash+@gold+@corr BankNetPositionUsd;END;
