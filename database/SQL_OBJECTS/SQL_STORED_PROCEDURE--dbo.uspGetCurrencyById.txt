CREATE OR ALTER PROCEDURE dbo.uspGetCurrencyById @CurrencyId uniqueidentifier AS SELECT CurrencyId,CurrencyCode,CurrencyNameEn,CurrencyNameAr,Symbol,DecimalPlaces,IsActive,CAST(CASE WHEN EXISTS(SELECT 1 FROM dbo.CorrespondentAccounts a WHERE a.CurrencyId=c.CurrencyId AND a.IsDeleted=0) THEN 1 ELSE 0 END AS bit) HasMovements FROM dbo.Currencies c WHERE CurrencyId=@CurrencyId AND IsDeleted=0;
