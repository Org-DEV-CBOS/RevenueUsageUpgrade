CREATE OR ALTER PROCEDURE dbo.uspAddResourceToCorrespondentAccount @ResourceDate date,@CorrespondentAccountId uniqueidentifier,@Amount decimal(19,4),@ResourceTypeId uniqueidentifier,@Notes nvarchar(300)=NULL,@CreatedBy nvarchar(100) AS BEGIN SET XACT_ABORT ON;BEGIN TRAN;UPDATE dbo.CorrespondentAccounts WITH(UPDLOCK) SET CurrentBalance=CurrentBalance+@Amount WHERE CorrespondentAccountId=@CorrespondentAccountId AND IsActive=1 AND IsDeleted=0;IF @@ROWCOUNT=0 THROW 52001,'Account not found or inactive.',1;INSERT dbo.Resources(ResourceDate,CorrespondentAccountId,Amount,ResourceTypeId,Notes,CreatedBy)VALUES(@ResourceDate,@CorrespondentAccountId,@Amount,@ResourceTypeId,@Notes,@CreatedBy);COMMIT;END;
