CREATE OR ALTER PROCEDURE dbo.uspDeleteCoverage @CoverageId uniqueidentifier,@DeletedBy nvarchar(100) AS BEGIN SET XACT_ABORT ON;BEGIN TRAN;DECLARE @f uniqueidentifier,@t uniqueidentifier,@v decimal(19,4);SELECT @f=FromCorrespondentAccountId,@t=ToCorrespondentAccountId,@v=Amount FROM dbo.Coverages WITH(UPDLOCK) WHERE CoverageId=@CoverageId AND DeletedTime IS NULL;IF @f IS NULL THROW 52022,'Coverage not found.',1;UPDATE dbo.CorrespondentAccounts SET CurrentBalance=CurrentBalance-@v WHERE CorrespondentAccountId=@t AND CurrentBalance>=@v;IF @@ROWCOUNT=0 THROW 52023,'Destination balance is insufficient to reverse coverage.',1;UPDATE dbo.CorrespondentAccounts SET CurrentBalance=CurrentBalance+@v WHERE CorrespondentAccountId=@f;UPDATE dbo.Coverages SET DeletedTime=SYSUTCDATETIME(),DeletedBy=@DeletedBy WHERE CoverageId=@CoverageId;COMMIT;END;
