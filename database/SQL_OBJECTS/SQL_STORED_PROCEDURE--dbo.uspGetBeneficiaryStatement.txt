CREATE OR ALTER PROCEDURE dbo.uspGetBeneficiaryStatement @BeneficiaryId uniqueidentifier,@StartDate date=NULL,@EndDate date=NULL AS SELECT t.TransferId,t.TransferDate,a.AccountNumber,c.CorrespondentNameEn AccountName,t.Amount,t.TransferStatus,t.ReferenceNo,t.Purpose,t.ConfirmedTime,t.RejectedTime,t.RejectReason FROM dbo.Transfers t JOIN dbo.CorrespondentAccounts a ON a.CorrespondentAccountId=t.CorrespondentAccountId JOIN dbo.Correspondents c ON c.CorrespondentId=a.CorrespondentId WHERE t.BeneficiaryId=@BeneficiaryId AND t.DeletedTime IS NULL AND(@StartDate IS NULL OR t.TransferDate>=@StartDate)AND(@EndDate IS NULL OR t.TransferDate<DATEADD(day,1,@EndDate))ORDER BY t.TransferDate;
