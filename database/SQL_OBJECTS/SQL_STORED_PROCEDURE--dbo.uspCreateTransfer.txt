CREATE OR ALTER PROCEDURE dbo.uspCreateTransfer
    @TransferDate datetime2,
    @BeneficiaryId uniqueidentifier,
    @CorrespondentAccountId uniqueidentifier,
    @Amount decimal(19, 4),
    @Purpose nvarchar(500) = NULL,
    @ReferenceNo nvarchar(100) = NULL,
    @StatementDate date = NULL,
    @OperationTypeId uniqueidentifier = NULL,
    @ResourceTypeId uniqueidentifier = NULL,
    @UsageTypeId uniqueidentifier = NULL,
    @BankId uniqueidentifier = NULL,
    @CreatedBy nvarchar(100),
    @TransferId uniqueidentifier OUTPUT
AS
BEGIN
    SET NOCOUNT ON;

    SET @TransferId = NEWID();

    IF @OperationTypeId IS NULL
        SELECT TOP (1) @OperationTypeId = OperationTypeId
        FROM dbo.OperationTypes WHERE OperationTypeCode = 'TRANSFER';

    IF @UsageTypeId IS NULL
        SELECT TOP (1) @UsageTypeId = UsageTypeId
        FROM dbo.UsageTypes WHERE UsageTypeCode = 'BENEFICIARY';

    IF @ResourceTypeId IS NULL
        SELECT TOP (1) @ResourceTypeId = ResourceTypeId
        FROM dbo.ResourceTypes WHERE IsActive = 1 AND IsDeleted = 0
        ORDER BY ResourceTypeCode;

    IF @BankId IS NULL
        SELECT TOP (1) @BankId = BankId
        FROM dbo.Banks WHERE IsActive = 1 AND IsDeleted = 0
        ORDER BY BankCode;

    -- Purpose, ReferenceNo and StatementDate are optional while the transfer is
    -- Pending; uspConfirmTransfer is what requires a reference and statement date.
    -- Blanks are stored as NULL so that check is a simple IS NULL test.
    INSERT dbo.Transfers (
        TransferId,
        TransferDate,
        BeneficiaryId,
        CorrespondentAccountId,
        Amount,
        Purpose,
        ReferenceNo,
        StatementDate,
        OperationTypeId,
        ResourceTypeId,
        UsageTypeId,
        BankId,
        CreatedBy)
    VALUES (
        @TransferId,
        @TransferDate,
        @BeneficiaryId,
        @CorrespondentAccountId,
        @Amount,
        NULLIF(LTRIM(RTRIM(@Purpose)), N''),
        NULLIF(LTRIM(RTRIM(@ReferenceNo)), N''),
        @StatementDate,
        @OperationTypeId,
        @ResourceTypeId,
        @UsageTypeId,
        @BankId,
        @CreatedBy);
END;
