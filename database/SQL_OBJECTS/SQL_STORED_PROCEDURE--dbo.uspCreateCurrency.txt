CREATE OR ALTER PROCEDURE dbo.uspCreateCurrency @CurrencyCode char(3),@CurrencyNameEn nvarchar(100),@CurrencyNameAr nvarchar(100)=NULL,@Symbol nvarchar(10)=NULL,@DecimalPlaces tinyint=2,@CreatedBy nvarchar(100)=NULL AS INSERT dbo.Currencies(CurrencyCode,CurrencyNameEn,CurrencyNameAr,Symbol,DecimalPlaces,CreatedBy) OUTPUT inserted.CurrencyId VALUES(UPPER(@CurrencyCode),@CurrencyNameEn,@CurrencyNameAr,@Symbol,@DecimalPlaces,@CreatedBy);
