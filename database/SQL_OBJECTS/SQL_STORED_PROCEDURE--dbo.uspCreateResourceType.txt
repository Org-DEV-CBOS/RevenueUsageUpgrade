CREATE PROCEDURE dbo.uspCreateResourceType @ResourceTypeCode nvarchar(50),@ResourceTypeNameEn nvarchar(100),@ResourceTypeNameAr nvarchar(100)=NULL,@CreatedBy nvarchar(100)=NULL AS INSERT dbo.ResourceTypes(ResourceTypeCode,ResourceTypeNameEn,ResourceTypeNameAr,CreatedBy) OUTPUT inserted.ResourceTypeId VALUES(@ResourceTypeCode,@ResourceTypeNameEn,@ResourceTypeNameAr,@CreatedBy);
