CREATE PROCEDURE dbo.uspCreateReserveSnapshot @ReserveDate date,@GoldValue decimal(19,4),@CashInHand decimal(19,4),@Deposits decimal(19,4),@Notes nvarchar(500)=NULL,@CreatedBy nvarchar(100) AS INSERT dbo.ReserveSnapshots(ReserveDate,GoldValue,CashInHand,Deposits,Notes,CreatedBy)OUTPUT inserted.ReserveSnapshotId VALUES(@ReserveDate,@GoldValue,@CashInHand,@Deposits,@Notes,@CreatedBy);
