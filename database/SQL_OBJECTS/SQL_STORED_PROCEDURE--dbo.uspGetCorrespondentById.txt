CREATE PROCEDURE dbo.uspGetCorrespondentById @CorrespondentId uniqueidentifier AS SELECT c.CorrespondentId,c.CorrespondentCode,c.CorrespondentNameEn,c.CorrespondentNameAr,c.CountryId,k.CountryNameEn,k.CountryNameAr,c.IsActive FROM dbo.Correspondents c LEFT JOIN dbo.Countries k ON k.CountryId=c.CountryId WHERE c.CorrespondentId=@CorrespondentId AND c.IsDeleted=0;
